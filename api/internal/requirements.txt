# Per-function requirements for api/internal/welcome-send.py.
#
# Vercel's Python runtime uses requirements.txt co-located with the
# function file as the dep manifest — keeping it function-scoped lets
# us avoid bundling the heavy root-level deps (Pillow, pytesseract,
# selectolax, curl_cffi, openai) that the welcome-send path never
# touches at request time. Bundle stays <20MB instead of >150MB.
#
# The runtime path imports:
#   • automation/newsletter/welcome_dispatch  → httpx
#   • automation/newsletter/build_issue       → pure Python (commentary,
#                                                favorites, segments are
#                                                pure Python; llm_*
#                                                modules lazy-import
#                                                openai and gracefully
#                                                no-op without it)
#   • automation/newsletter/send              → httpx
#   • automation/newsletter/render_html       → pure Python (html.escape)
#   • automation/posthog_client               → posthog (optional;
#                                                silent no-op without it)
#
# Pinning to the same versions the root requirements.txt uses so dev
# and Vercel runtime behave identically. Bump in lockstep.
httpx==0.27.2
posthog>=3.0,<4.0
# --- Vercel Python runtime pin (workaround, 2026-07-15) ---
# vercel_runtime 0.16.0 ships a broken .dist-info/licenses/ that copies its
# own src tree; Vercel's build-time bytecode compile then dies with
# `ENOENT ... vercel_runtime-0.16.0.dist-info/licenses/.../__pycache__/__init__.cpython-312.pyc`,
# failing EVERY Python-function deploy (prod + preview) regardless of cache
# or tree size. Confirmed on a clean 400MB no-cache build (1m13s fail).
# 0.15.0 is the last good release. Vercel Support case: 01QGyoEPWLy3yBKc.
# Remove once Vercel patches the builder.
vercel_runtime==0.15.0
